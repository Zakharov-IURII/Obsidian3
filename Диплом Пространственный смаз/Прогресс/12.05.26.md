``` python
def load_image(filepath):
    """читает файл с диска и возвращает две картинки - RGB и grayscale."""
# превращение строки пути в удобный формат
    filepath = Path(filepath)

# imread требует строку
    img = cv2.imread(str(filepath))

    if img is None:
        print(f"Ошибка: не удалось прочитать {filepath}")
        return None
    
# конвертация из BGR в RGB
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
# конвертаця в grayscale (теперь img_gray.shape вернет два числа, а не три)
    img_gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

    return img_rgb, img_gray
```

- Добавлена утилита fast_fourier_transform:
``` python
def fast_fourier_transform(img):
    """Возвращает комплексный спектр (любого изображения) и его модуль"""
    img_fft = np.fft.fft2(img)
    img_fft_shifted = np.fft.fftshift(img_fft)
    magnitude = np.log(np.abs(img_fft_shifted) + 1)
    return img_fft, magnitude
```

- Задание ядра (функции рассеяния точки - PSF) линейного горизонтального смаза:

Сумма всех чисел в матрице должны быть равна 1 для сохранения общей яркости изображения (>1 - станет светлее, <1 - темнее) - нормировка;
Размер ядра = размеру изображения: перемножение матриц в частотной области;

*cv2.filter2D()* - двумерная пространственная свертка, реализующая разные эффекты, в том числе пространственный смаз. Она не создает ядро, а применяет то, которое ей дали, и не следит за нормализацией.

Ядро будет сделано вручную:
Матрица той же размерности, что и изображение;
Заполнена нулями;
Точно в центре - симметричная (для простоты) относительно центра горизонтальная линия из ненулевых равных значений, в сумме дающих 1;
Длина смаза (длина линии) - для симметрии относительно центра выбираем нечетное число.

``` python
# длина смаза (на сколько пикселей "размазывается" каждый)

center_row = rows // 2

half = smear_length // 2

# нули на позициях заменяются на 1/smear_lenght,
# center_row - центральная строка изображения,
# center_col - центральный столбец,
# half - чтобы от центрального столбца вправо и влево по половине длины смаза отступить
# +1 - чтобы срез захватывал конец диапазона (иначе не включит)
kernel[center_row, center_col - half : center_col + half + 1] = 1 / smear_length

# БПФ ядра
kernel_fft, magnitude_kernel = fast_fourier_transform(kernel)

# БПФ изображения

blurred_fft_shifted = np.fft.fftshift(blurred_fft)
```
