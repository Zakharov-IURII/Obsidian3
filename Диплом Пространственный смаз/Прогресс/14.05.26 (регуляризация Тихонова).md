``` python
def fast_blur(img, smear_length):
    """Возвращает изображение с горизонтальным смазом заданной длины"""
    kernel_small = np.ones((1, smear_length)) / smear_length
    blurred = cv2.filter2D(img, -1, kernel_small)
    return blurred

def create_kernel(img, smear_length):
    """Возвращает PSF горизонтального смаза для восстановления в частотной области"""
    rows, cols = img.shape
    kernel = np.ones((1, smear_length)) / smear_length
    center_row = rows // 2
    center_col = cols // 2
    half = smear_length // 2
    kernel[center_row, center_col - half: center_col + half + 1] = 1 / smear_length
    
    # подготовка к БПФ с помощью ifftshift - перемещение самого ядра в (0, 0)
    kernel = np.fft.ifftshift(kernel)

    return kernel
```
