``` python
def load_image(filepath):
    """читает файл с диска и возвращает две картинки - RGB и grayscale."""
# превращение строки пути в удобный формат
    filepath = Path(filepath)

# imread требует строку
    img = cv2.imread(str(filepath))

    if img is None:
        print(f"Ошибка: не удалось прочитать {filepath}")
        return None
    
# конвертация из BGR в RGB и сразу перевод в float32 для лучшей точности
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB).astype(np.float32)
# конвертаця в grayscale
    img_gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY).astype(np.float32)

    return img_rgb, img_gray

```

#### Восстановление после смаза
1. Первый путь - обратная фильтрация в лоб
Взять спектр смазанного изображения и поделить на спектр ядра смаза.

Поскольку смаз теперь делаем с помощью cv2.filter2D и подаваемого ему на вход компактного ядра (1 на 25, нормированного), 
но умножение/деление в частотной области требуют равных размеров операндов-массивов, 
сделаем два ядра - компактное и большое, дополненное нулями и <u>с правильным центрированием</u> 
(cv2.filter2D считает центр ядра опорной точкой, нам нужно разместить наше ядро так, чтобы центр точно был на координатах (0,0) исходного, иначе будет <u>линейный фазовый сдвиг</u> - возможны сдвиги/завороты изображения.
Для этого можно гибко применить ifftshift - не ОБПФ, а операция, обратная fftshift: она корректно разместит центр массива в левом верхнем углу, а левую часть ядра ).

``` python
img_rgb,img_gray = load_image(r"C:\Users\user\Projects\Diplom\data\car.jpg")

# PSF (компактное ядро - только для получения смаза через filter2D - для дальнейшего 
# восстановления нужно ядро с размерами изображения, 
# чтобы корректно поделить в частотной области)
smear_length = 25
kernel_small = np.ones((1, smear_length)) / smear_length

# Смаз
blurred_img = cv2.filter2D(img_gray, -1, kernel_small)
show_image(blurred_img)

# PSF - большое ядро для деления в частотной области
rows, cols = img_gray.shape
kernel = np.zeros((rows, cols), dtype=np.float32)
center_row = rows // 2
center_col = cols // 2
half = smear_length // 2
kernel[center_row, center_col - half: center_col + half + 1] = 1 / smear_length

# разместили ядро строго в центре - теперь сместим его центр тяжести в (0,0) с помощью
# ifftshift - упакуем для последующего fft2
kernel = np.fft.ifftshift(kernel)

# теперь центр ядра лежит в верхнем левом углу, 
# левая его часть перенеслась в верхний правый угол
# Спектр:
kernel_fft = np.fft.fft2(kernel)

# Инверсная фильтрация:
blurred_fft = np.fft.fft2(blurred_img)
img_restored_fft = blurred_fft / kernel_fft
magnitude = np.log(np.abs(np.fft.fftshift(img_restored_fft)) + 1)
show_image(magnitude)

# Само восстановленное изображение:
img_restored = np.real(np.fft.ifft2(img_restored_fft))
show_image(img_restored)
```
